local function Icon(name, size, parent, color)
	local img = Instance.new("ImageLabel")
	img.Name = name
	img.Size = UDim2.fromOffset(size or 20, size or 20)
	img.BackgroundTransparency = 1
	img.Image = Icons[name] or Icons["star"]
	img.ImageColor3 = color or Color3.fromRGB(200, 200, 210)
	img.Parent = parent
	return img
end

local function Hex(hex)
	hex = hex:gsub("#", "")
	return Color3.fromRGB(
		tonumber(hex:sub(1, 2), 16),
		tonumber(hex:sub(3, 4), 16),
		tonumber(hex:sub(5, 6), 16)
	)
end

local function New(class, props, parent)
	local inst = Instance.new(class)
	for k, v in pairs(props) do inst[k] = v end
	inst.Parent = parent
	return inst
end

local function Round(obj, r)
	New("UICorner", { CornerRadius = UDim.new(0, r or 10) }, obj)
end

local function Tween(obj, time, props, style, dir)
	local t = TweenService:Create(
		obj,
		TweenInfo.new(time or 0.2, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
		props
	)
	t:Play()
	return t
end

local CurrentTheme = Celestial.Themes[1]
local ThemeChanged = Instance.new("BindableEvent")

local function TC(which) return Hex(CurrentTheme[which]) end
local function ThemeGradient(obj)
	local g = New("UIGradient", { Rotation = 15 }, obj)
	local function apply()
		g.Color = ColorSequence.new(TC("C1"), TC("C2"))
	end
	apply()
	ThemeChanged.Event:Connect(apply)
	return g
end

	local visible = true

		Size = UDim2.fromOffset(740, 600),
		Position = UDim2.fromScale(0.5, 0.5),
		AnchorPoint = Vector2.new(0.5, 0.5),
		BackgroundColor3 = Color3.fromRGB(13, 13, 16),
		BorderSizePixel = 0,
		Active = true,
		GroupTransparency = 0,
	}, gui)
	Round(Main, 12)

	New("UIStroke", { Color = Color3.fromRGB(45, 45, 55), Thickness = 1 }, Main)

	-- Анимация появления окна при запуске
	Main.Size = UDim2.fromOffset(700, 560)
	Tween(Main, 0.35, { Size = UDim2.fromOffset(740, 600) }, Enum.EasingStyle.Back)

	UserInputService.InputBegan:Connect(function(input, processed)
		if processed then return end
		if input.KeyCode == Enum.KeyCode.RightShift then
			visible = not visible
			if visible then
				gui.Enabled = true
				Main.GroupTransparency = 1
				Main.Position = UDim2.fromScale(0.5, 0.53)
				Tween(Main, 0.25, { GroupTransparency = 0, Position = UDim2.fromScale(0.5, 0.5) })
			else
				Tween(Main, 0.25, { GroupTransparency = 1, Position = UDim2.fromScale(0.5, 0.53) })
				task.delay(0.26, function() gui.Enabled = false end)
			end
		end
	end)

	-- Перетаскивание
	do
		local dragging, dragStart, startPos
		Main.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 then
				dragging, dragStart, startPos = true, input.Position, Main.Position
			end
		end)
		UserInputService.InputEnded:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
		end)
		UserInputService.InputChanged:Connect(function(input)
			if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
				local d = input.Position - dragStart
				Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
			end
		end)
	end

	function Window:CreateTab(name, iconName, isExtra)
		local holder = isExtra and TabsExtra or TabsMain

		local btn = New("TextButton", {
			Size = UDim2.new(1, 0, 0, 40),
			BackgroundColor3 = Color3.fromRGB(19, 19, 24),
			Text = "", AutoButtonColor = false, BorderSizePixel = 0,
		}, holder)
		Round(btn, 10)
		btn.ClipsDescendants = true

		local tabIcon = Icon(iconName or "star", 20, btn)
		tabIcon.Position = UDim2.fromOffset(14, 10)

		local lbl = New("TextLabel", {
			Size = UDim2.new(1, -50, 1, 0), Position = UDim2.fromOffset(46, 0),
			BackgroundTransparency = 1, Text = name,
			TextColor3 = Color3.fromRGB(165, 165, 180), Font = Enum.Font.GothamBold,
			TextSize = 15, TextXAlignment = Enum.TextXAlignment.Left,
		}, btn)

		-- Страница вкладки
		local page = New("ScrollingFrame", {
			Size = UDim2.new(1, -30, 1, -72), Position = UDim2.fromOffset(15, 64),
			BackgroundTransparency = 1, BorderSizePixel = 0,
			CanvasSize = UDim2.new(), ScrollBarThickness = 3,
			ScrollBarImageColor3 = Color3.fromRGB(70, 70, 85),
			Visible = false,
		}, Content)
		local grid = New("UIGridLayout", {
			CellSize = UDim2.fromOffset(250, 72), CellPadding = UDim2.fromOffset(10, 10),
			SortOrder = Enum.SortOrder.LayoutOrder,
		}, page)

		local Tab = {
			Name = name,
			_Btn = btn, _Icon = tabIcon, _Lbl = lbl,
			_Page = page, _Grid = grid, _Cards = {},
			_IsThemes = false,
		}
		table.insert(AllTabs, Tab)

		-- ===== ХОВЕР НА КНОПКЕ РАЗДЕЛА: бегущий блик + подсветка =====
		btn.MouseEnter:Connect(function()
			if Tab._Active then return end
			Tween(btn, 0.2, { BackgroundColor3 = Color3.fromRGB(28, 28, 36) })
			Tween(tabIcon, 0.2, { ImageColor3 = TC("C1") })

			local sweep = New("Frame", {
				Size = UDim2.fromOffset(60, 200),
				Position = UDim2.fromOffset(-60, 0),
				BackgroundColor3 = Color3.new(1, 1, 1),
				BackgroundTransparency = 0.93,
				ZIndex = 5, BorderSizePixel = 0,
			}, btn)
			Round(sweep, 30)
			local sg = ThemeGradient(sweep)
			sg.Rotation = 90
			Tween(sweep, 0.45, { Position = UDim2.fromOffset(250, 0) })
			task.delay(0.5, function() sweep:Destroy() end)
		end)
		btn.MouseLeave:Connect(function()
			if Tab._Active then return end
			Tween(btn, 0.2, { BackgroundColor3 = Color3.fromRGB(19, 19, 24) })
			Tween(tabIcon, 0.2, { ImageColor3 = Color3.fromRGB(200, 200, 210) })
		end)

		-- ===== ВЫБОР ВКЛАДКИ =====
		btn.MouseButton1Click:Connect(function()
			for _, t in pairs(AllTabs) do
				t._Active = false
				t._Page.Visible = false
				t._Btn.BackgroundTransparency = 0
				t._Btn.BackgroundColor3 = Color3.fromRGB(19, 19, 24)
				for _, c in pairs(t._Btn:GetChildren()) do
					if c:IsA("UIGradient") then c:Destroy() end
				end
				Tween(t._Lbl, 0.2, { TextColor3 = Color3.fromRGB(165, 165, 180) })
				Tween(t._Icon, 0.2, { ImageColor3 = Color3.fromRGB(200, 200, 210) })
			end

			page.Visible = true

			-- Градиентная кнопка активной вкладки
			btn.BackgroundColor3 = Color3.new(1, 1, 1)
			ThemeGradient(btn)
			lbl.TextColor3 = Color3.new(1, 1, 1)
			tabIcon.ImageColor3 = Color3.new(1, 1, 1)
			tabIcon.ZIndex = 3
			btn.ZIndex = 1

				if c:IsA("ImageLabel") or c:IsA("TextLabel") then c.ZIndex = 3 end
			end

			page.Position = UDim2.fromOffset(15, 74)
			Tween(page, 0.3, { Position = UDim2.fromOffset(15, 64) }, Enum.EasingStyle.Quint)
		end)

		-- ===== ПОИСК =====
		function Tab._Refresh(query)
			if Tab._IsThemes then
				for _, c in pairs(Tab._Cards) do
					c.Card.Visible = query == "" or string.find(string.lower(c.Name), query, 1, true) ~= nil
				end
			else
				for _, c in pairs(Tab._Cards) do
					c.Card.Visible = query == "" or string.find(string.lower(c.Title .. " " .. c.Desc), query, 1, true) ~= nil
				end
			end
		end

		function Tab:AddCard(title, desc, callback, iconName)
			local card = New("TextButton", {
				BackgroundColor3 = Color3.fromRGB(19, 19, 24),
				Text = "", AutoButtonColor = false, BorderSizePixel = 0,
			}, page)
			Round(card, 10)
			card.ClipsDescendants = true

			local stroke = New("UIStroke", { Color = Color3.fromRGB(34, 34, 44), Thickness = 1 }, card)

			-- Градиентная заливка включённого состояния
			local glow = New("Frame", {
				Size = UDim2.fromScale(1, 1), BackgroundColor3 = Color3.new(1, 1, 1),
				Visible = false, ZIndex = 0, BorderSizePixel = 0,
			}, card)
			Round(glow, 10)
			ThemeGradient(glow)

			Icon(iconName or "star", 18, card, Color3.fromRGB(150, 150, 165)).Position = UDim2.new(1, -30, 0, 10)

			New("TextLabel", {
				Size = UDim2.new(1, -62, 0, 22), Position = UDim2.fromOffset(12, 8),
				BackgroundTransparency = 1, Text = title,
				TextColor3 = Color3.new(1, 1, 1), Font = Enum.Font.GothamBold,
				TextSize = 15, TextXAlignment = Enum.TextXAlignment.Left, ZIndex = 3,
			}, card)
			New("TextLabel", {
				Size = UDim2.new(1, -24, 0, 32), Position = UDim2.fromOffset(12, 32),
				BackgroundTransparency = 1, Text = desc or "",
				TextColor3 = Color3.fromRGB(105, 105, 120), TextWrapped = true,
				Font = Enum.Font.Gotham, TextSize = 11,
				TextXAlignment = Enum.TextXAlignment.Left,
				TextYAlignment = Enum.TextYAlignment.Top, ZIndex = 3,
			}, card)

			local enabled = false
			local state = {
				Title = title, Desc = desc or "", Card = card,
			}
			table.insert(Tab._Cards, state)

			local function Set(on, silent)
				enabled = on
				glow.Visible = on
				Tween(glow, 0.25, { BackgroundTransparency = on and 0.62 or 1 })
				stroke.Color = on and TC("C1") or Color3.fromRGB(34, 34, 44)
				stroke.Transparency = on and 0.3 or 0
				if on and not silent then
					-- пульс при клике
					local pulse = New("Frame", {
						Size = UDim2.fromScale(1, 1), BackgroundColor3 = Color3.new(1, 1, 1),
						BackgroundTransparency = 0.7, ZIndex = 4, BorderSizePixel = 0,
					}, card)
					Round(pulse, 10)
					Tween(pulse, 0.4, { BackgroundTransparency = 1 })
					task.delay(0.42, function() pulse:Destroy() end)
				end
				if callback then callback(on) end
			end
			state.SetState = Set
			state.GetState = function() return enabled end

			-- ===== ХОВЕР НА КАРТОЧКЕ =====
			card.MouseEnter:Connect(function()
				Tween(card, 0.18, { BackgroundColor3 = Color3.fromRGB(26, 26, 34) })
				Tween(card, 0.18, { Size = UDim2.fromOffset(255, 76) }, Enum.EasingStyle.Back)
				if not enabled then
					Tween(stroke, 0.18, { Color = TC("C1"), Transparency = 0.55 })
				end
			end)
			card.MouseLeave:Connect(function()
				Tween(card, 0.18, { BackgroundColor3 = Color3.fromRGB(19, 19, 24), Size = UDim2.fromOffset(250, 72) })
				if not enabled then
					Tween(stroke, 0.18, { Color = Color3.fromRGB(34, 34, 44), Transparency = 0 })
				end
			end)

			card.MouseButton1Click:Connect(function()
				Set(not enabled)
			end)

			return state
		end

		for i, theme in ipairs(Celestial.Themes) do
			local card = New("TextButton", {
				BackgroundColor3 = Color3.new(1, 1, 1),
				Text = "", AutoButtonColor = false, BorderSizePixel = 0,
			}, themesTab._Page)
			Round(card, 10)
			local g = New("UIGradient", {
				Color = ColorSequence.new(Hex(theme.C1), Hex(theme.C2)),
				Rotation = 15,
			}, card)

			New("TextLabel", {
				Size = UDim2.new(1, -12, 0, 20), Position = UDim2.fromOffset(9, 7),
				BackgroundTransparency = 1, Text = theme.Name,
				TextColor3 = Color3.new(1, 1, 1), Font = Enum.Font.GothamBold,
				TextSize = 12, TextXAlignment = Enum.TextXAlignment.Left,
				TextTruncate = Enum.TextTruncate.AtEnd,
			}, card)
			New("TextLabel", {
				Size = UDim2.new(1, -12, 0, 30), Position = UDim2.fromOffset(9, 27),
				BackgroundTransparency = 1, Text = theme.C1 .. "\n" .. theme.C2,
				TextColor3 = Color3.new(1, 1, 1), TextTransparency = 0.3,
				Font = Enum.Font.Code, TextSize = 10,
				TextXAlignment = Enum.TextXAlignment.Left,
				TextYAlignment = Enum.TextYAlignment.Top,
			}, card)

			local num = New("TextLabel", {
				Size = UDim2.fromOffset(22, 22), AnchorPoint = Vector2.new(1, 1),
				Position = UDim2.new(1, -8, 1, -8),
				BackgroundColor3 = Color3.new(1, 1, 1), BackgroundTransparency = 0.25,
				Text = tostring(i), TextColor3 = Color3.new(0, 0, 0),
				Font = Enum.Font.GothamBold, TextSize = 12,
			}, card)
			Round(num, 7)

			state = { Name = theme.Name, Card = card }
			table.insert(themesTab._Cards, state)

			-- ховер
			card.MouseEnter:Connect(function()
				Tween(card, 0.15, { Size = UDim2.fromOffset(164, 81) }, Enum.EasingStyle.Back)
			end)
			card.MouseLeave:Connect(function()
				Tween(card, 0.15, { Size = UDim2.fromOffset(160, 78) })
			end)

			-- выбор темы
			card.MouseButton1Click:Connect(function()
				CurrentTheme = theme

				if selectedStroke then selectedStroke:Destroy() end
				selectedStroke = New("UIStroke", {
					Color = Color3.new(1, 1, 1), Thickness = 2.5,
				}, card)

				ThemeChanged:Fire()

				-- перекрашиваем активную кнопку вкладки
				for _, t in pairs(AllTabs) do
					if t._Active then
						for _, c in pairs(t._Btn:GetChildren()) do
							if c:IsA("UIGradient") then
								c.Color = ColorSequence.new(TC("C1"), TC("C2"))
							end
						end
					end
					-- обновляем включённые карточки
					for _, c in pairs(t._Cards) do
						if c.SetState and c.GetState and c.GetState() then
							c.SetState(true, true)
						end
					end
				end
			end)

			if i == 1 then
				selectedStroke = New("UIStroke", { Color = Color3.new(1, 1, 1), Thickness = 2.5 }, card)
			end
		end
	end

	-- Поиск в реальном времени
	SearchBox:GetPropertyChangedSignal("Text"):Connect(function()
		local q = string.lower(SearchBox.Text)
		for _, t in pairs(AllTabs) do
			t._Refresh(q)
		end
	end)

	Window.Gui = gui
	Window.Main = Main
	Window:SetupThemesTab()

	-- Автооткрытие первой основной вкладки
	if AllTabs[1] then
		AllTabs[1]._Btn:FindFirstChildOfClass("TextLabel").TextColor3 = Color3.fromRGB(165, 165, 180)
	end

	return Window
end

return Celestial
