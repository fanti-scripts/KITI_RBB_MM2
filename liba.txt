--// ================== ЗАГРУЗКА ИКОНОК ==================
local ICON_REPO = "https://raw.githubusercontent.com/ActualSkelet/Roblox-Icons/main/icons/%s.png"
local cacheFolder = "oxlib_icons"

if not isfolder(cacheFolder) then makefolder(cacheFolder) end

local function GetIcon(name)
	local path = cacheFolder .. "/" .. name .. ".png"
	if not isfile(path) then
		local ok, data = pcall(game.HttpGet, game, ICON_REPO:format(name))
		if ok then writefile(path, data) end
	end
	if isfile(path) then
		if getcustomasset then
			return getcustomasset(path)
		elseif syn and syn.is_cached_asset then
			return syn.get_custom_asset(path)
		end
	end
	return "rbxassetid://0" -- заглушка
end

--// ================== УТИЛИТЫ ==================
local function Create(class, props, parent)
	local inst = Instance.new(class)
	for k, v in pairs(props) do
		inst[k] = v
	end
	inst.Parent = parent
	return inst
end

local function Tween(obj, time, props, style, dir)
	TweenService:Create(obj, TweenInfo.new(time, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out), props):Play()
end

local function Corner(parent, radius)
	return Create("UICorner", { CornerRadius = UDim.new(0, radius or 8) }, parent)
end

--// ================== ОКНО ==================
function OxLib:CreateWindow(config)
	config = config or {}
	local Title = config.Title or "OxLib"
	local Theme = config.Theme or {
		Background = Color3.fromRGB(18, 18, 24),
		Secondary = Color3.fromRGB(28, 28, 38),
		Accent = Color3.fromRGB(120, 90, 255),
		Text = Color3.fromRGB(240, 240, 245),
		SubText = Color3.fromRGB(150, 150, 165),
	}

	local ScreenGui = Create("ScreenGui", {
		Name = "OxLib",
		ResetOnSpawn = false,
		ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
	}, game.CoreGui)

	local Main = Create("Frame", {
		Size = UDim2.fromOffset(560, 380),
		Position = UDim2.new(0.5, -280, 0.5, -190),
		BackgroundColor3 = Theme.Background,
		BorderSizePixel = 0,
		Active = true,
		ClipsDescendants = true,
	}, ScreenGui)
	Corner(Main, 12)

	-- анимация появления окна
	Main.Size = UDim2.fromOffset(0, 0)
	Tween(Main, 0.4, { Size = UDim2.fromOffset(560, 380) })

	-- перетаскивание
	do
		local dragging, dragStart, startPos
		Main.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 then
				dragging = true
				dragStart = input.Position
				startPos = Main.Position
			end
		end)
		UserInputService.InputChanged:Connect(function(input)
			if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
				local delta = input.Position - dragStart
				Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
			end
		end)
		UserInputService.InputEnded:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
		end)
	end

	-- шапка
	local TopBar = Create("Frame", {
		Size = UDim2.new(1, 0, 0, 42),
		BackgroundColor3 = Theme.Secondary,
		BorderSizePixel = 0,
	}, Main)
	Corner(TopBar, 12)

	Create("ImageLabel", { -- иконка окна
		Size = UDim2.fromOffset(22, 22),
		Position = UDim2.new(0, 12, 0.5, -11),
		BackgroundTransparency = 1,
		Image = GetIcon("lightning-bolt"),
		ImageColor3 = Theme.Accent,
	}, TopBar)

	Create("TextLabel", {
		Size = UDim2.new(1, -100, 1, 0),
		Position = UDim2.new(0, 44, 0, 0),
		BackgroundTransparency = 1,
		Text = Title,
		TextColor3 = Theme.Text,
		TextSize = 16,
		Font = Enum.Font.GothamBold,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, TopBar)

	-- кнопка свернуть
	local MinBtn = Create("ImageButton", {
		Size = UDim2.fromOffset(26, 26),
		Position = UDim2.new(1, -64, 0.5, -13),
		BackgroundTransparency = 1,
		Image = GetIcon("minus"),
		ImageColor3 = Theme.SubText,
	}, TopBar)

	-- кнопка закрыть
	Create("ImageButton", {
		Size = UDim2.fromOffset(26, 26),
		Position = UDim2.new(1, -34, 0.5, -13),
		BackgroundTransparency = 1,
		Image = GetIcon("x"),
		ImageColor3 = Theme.SubText,
	}, TopBar).MouseButton1Click:Connect(function()
		Tween(Main, 0.25, { Size = UDim2.fromOffset(0, 0) })
		task.wait(0.25)
		ScreenGui:Destroy()
	end)

	MinBtn.MouseButton1Click:Connect(function()
		local minimized = Main.Size ~= UDim2.fromOffset(560, 380)
		Tween(Main, 0.35, { Size = minimized and UDim2.fromOffset(560, 380) or UDim2.fromOffset(560, 42) })
		Tween(MinBtn, 0.2, { Rotation = minimized and 0 or 180 })
	end)

	-- панель вкладок
	local TabHolder = Create("ScrollingFrame", {
		Size = UDim2.new(0, 140, 1, -52),
		Position = UDim2.new(0, 6, 0, 46),
		BackgroundTransparency = 1,
		ScrollBarThickness = 0,
		CanvasSize = UDim2.new(),
		AutomaticCanvasSize = Enum.AutomaticSize.Y,
	}, Main)
	Create("UIListLayout", { Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder }, TabHolder)

	local PageHolder = Create("Frame", {
		Size = UDim2.new(1, -158, 1, -52),
		Position = UDim2.new(0, 152, 0, 46),
		BackgroundTransparency = 1,
		ClipsDescendants = true,
	}, Main)

	local Window = { Theme = Theme }
	local firstTab = true

	function Window:CreateTab(tabName, iconName)
		local TabBtn = Create("TextButton", {
			Size = UDim2.new(1, 0, 0, 36),
			BackgroundColor3 = Theme.Background,
			BackgroundTransparency = 1,
			Text = "",
			AutoButtonColor = false,
		}, TabHolder)
		Corner(TabBtn, 8)

		Create("ImageLabel", {
			Size = UDim2.fromOffset(18, 18),
			Position = UDim2.new(0, 10, 0.5, -9),
			BackgroundTransparency = 1,
			Image = GetIcon(iconName or "circle-star"),
			ImageColor3 = Theme.SubText,
		}, TabBtn)

		Create("TextLabel", {
			Size = UDim2.new(1, -40, 1, 0),
			Position = UDim2.new(0, 36, 0, 0),
			BackgroundTransparency = 1,
			Text = tabName,
			TextColor3 = Theme.SubText,
			TextSize = 14,
			Font = Enum.Font.GothamMedium,
			TextXAlignment = Enum.TextXAlignment.Left,
		}, TabBtn)

		local Page = Create("ScrollingFrame", {
			Size = UDim2.new(1, 0, 1, 0),
			BackgroundTransparency = 1,
			Visible = false,
			ScrollBarThickness = 3,
			ScrollBarImageColor3 = Theme.Accent,
			CanvasSize = UDim2.new(),
			AutomaticCanvasSize = Enum.AutomaticSize.Y,
		}, PageHolder)
		Create("UIListLayout", { Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder }, Page)
		Create("UIPadding", { PaddingLeft = UDim.new(0, 4), PaddingRight = UDim.new(0, 8), PaddingTop = UDim.new(0, 4) }, Page)

		local selected = false
		local function Select()
			for _, other in ipairs(PageHolder:GetChildren()) do
				if other:IsA("ScrollingFrame") then other.Visible = false end
			end
			for _, btn in ipairs(TabHolder:GetChildren()) do
				if btn:IsA("TextButton") then
					Tween(btn, 0.2, { BackgroundTransparency = 1 })
					Tween(btn:FindFirstChildOfClass("ImageLabel"), 0.2, { ImageColor3 = Theme.SubText })
					Tween(btn:FindFirstChildOfClass("TextLabel"), 0.2, { TextColor3 = Theme.SubText })
				end
			end
			Page.Visible = true
			Page.Size = UDim2.new(1, 0, 1, 20)
			Tween(Page, 0.3, { Size = UDim2.new(1, 0, 1, 0) }) -- анимация страницы
			Tween(TabBtn, 0.2, { BackgroundTransparency = 0, BackgroundColor3 = Theme.Secondary })
			Tween(TabBtn:FindFirstChildOfClass("ImageLabel"), 0.2, { ImageColor3 = Theme.Accent })
			Tween(TabBtn:FindFirstChildOfClass("TextLabel"), 0.2, { TextColor3 = Theme.Text })
		end

		TabBtn.MouseButton1Click:Connect(Select)
		if firstTab then firstTab = false; task.defer(Select) end

		-- ховер на кнопку вкладки
		TabBtn.MouseEnter:Connect(function()
			if not selected then Tween(TabBtn, 0.15, { BackgroundTransparency = 0.7 }) end
		end)
		TabBtn.MouseLeave:Connect(function()
			if not selected then Tween(TabBtn, 0.15, { BackgroundTransparency = 1 }) end
		end)

		local Tab = {}

		-- ---------- СЕКЦИЯ ----------
		function Tab:AddSection(text)
			Create("TextLabel", {
				Size = UDim2.new(1, 0, 0, 24),
				BackgroundTransparency = 1,
				Text = text,
				TextColor3 = Theme.Accent,
				TextSize = 13,
				Font = Enum.Font.GothamBold,
				TextXAlignment = Enum.TextXAlignment.Left,
			}, Page)
		end

		-- ---------- КНОПКА ----------
		function Tab:AddButton(text, iconName, callback)
			callback = callback or function() end
			local Btn = Create("TextButton", {
				Size = UDim2.new(1, 0, 0, 38),
				BackgroundColor3 = Theme.Secondary,
				Text = "",
				AutoButtonColor = false,
			}, Page)
			Corner(Btn, 8)

			Create("ImageLabel", {
				Size = UDim2.fromOffset(16, 16),
				Position = UDim2.new(0, 12, 0.5, -8),
				BackgroundTransparency = 1,
				Image = GetIcon(iconName or "play-small"),
				ImageColor3 = Theme.Text,
			}, Btn)

			Create("TextLabel", {
				Size = UDim2.new(1, -50, 1, 0),
				Position = UDim2.new(0, 38, 0, 0),
				BackgroundTransparency = 1,
				Text = text,
				TextColor3 = Theme.Text,
				TextSize = 14,
				Font = Enum.Font.GothamMedium,
				TextXAlignment = Enum.TextXAlignment.Left,
			}, Btn)

			Btn.MouseEnter:Connect(function() Tween(Btn, 0.15, { BackgroundColor3 = Theme.Accent }) end)
			Btn.MouseLeave:Connect(function() Tween(Btn, 0.15, { BackgroundColor3 = Theme.Secondary }) end)
			Btn.MouseButton1Click:Connect(function()
				Tween(Btn, 0.08, { Size = UDim2.new(1, -6, 0, 34) })
				task.wait(0.08)
				Tween(Btn, 0.15, { Size = UDim2.new(1, 0, 0, 38) })
				callback()
			end)
		end

		-- ---------- ТУМБЛЕР ----------
		function Tab:AddToggle(text, default, callback)
			default = default or false
			callback = callback or function() end
			local state = default

			local Holder = Create("TextButton", {
				Size = UDim2.new(1, 0, 0, 38),
				BackgroundColor3 = Theme.Secondary,
				Text = "",
				AutoButtonColor = false,
			}, Page)
			Corner(Holder, 8)

			Create("TextLabel", {
				Size = UDim2.new(1, -70, 1, 0),
				Position = UDim2.new(0, 12, 0, 0),
				BackgroundTransparency = 1,
				Text = text,
				TextColor3 = Theme.Text,
				TextSize = 14,
				Font = Enum.Font.GothamMedium,
				TextXAlignment = Enum.TextXAlignment.Left,
			}, Holder)

			local Switch = Create("Frame", {
				Size = UDim2.fromOffset(40, 22),
				Position = UDim2.new(1, -52, 0.5, -11),
				BackgroundColor3 = state and Theme.Accent or Color3.fromRGB(60, 60, 75),
			}, Holder)
			Corner(Switch, 11)

			local Knob = Create("Frame", {
				Size = UDim2.fromOffset(16, 16),
				Position = state and UDim2.new(1, -19, 0.5, -8) or UDim2.new(0, 3, 0.5, -8),
				BackgroundColor3 = Color3.fromRGB(255, 255, 255),
			}, Switch)
			Corner(Knob, 8)

			Holder.MouseButton1Click:Connect(function()
				state = not state
				Tween(Switch, 0.25, { BackgroundColor3 = state and Theme.Accent or Color3.fromRGB(60, 60, 75) })
				Tween(Knob, 0.25, { Position = state and UDim2.new(1, -19, 0.5, -8) or UDim2.new(0, 3, 0.5, -8) }, Enum.EasingStyle.Back)
				callback(state)
			end)
		end

		-- ---------- СЛАЙДЕР ----------
		function Tab:AddSlider(text, min, max, default, callback)
			min, max, default = min or 0, max or 100, default or min
			callback = callback or function() end

			local Holder = Create("Frame", {
				Size = UDim2.new(1, 0, 0, 56),
				BackgroundColor3 = Theme.Secondary,
			}, Page)
			Corner(Holder, 8)

			local ValueLabel
			Create("TextLabel", {
				Size = UDim2.new(1, -80, 0, 24),
				Position = UDim2.new(0, 12, 0, 2),
				BackgroundTransparency = 1,
				Text = text,
				TextColor3 = Theme.Text,
				TextSize = 14,
				Font = Enum.Font.GothamMedium,
				TextXAlignment = Enum.TextXAlignment.Left,
			}, Holder)
			ValueLabel = Create("TextLabel", {
				Size = UDim2.new(0, 60, 0, 24),
				Position = UDim2.new(1, -68, 0, 2),
				BackgroundTransparency = 1,
				Text = tostring(default),
				TextColor3 = Theme.Accent,
				TextSize = 13,
				Font = Enum.Font.GothamBold,
				TextXAlignment = Enum.TextXAlignment.Right,
			}, Holder)

			local Bar = Create("Frame", {
				Size = UDim2.new(1, -24, 0, 6),
				Position = UDim2.new(0, 12, 0, 36),
				BackgroundColor3 = Color3.fromRGB(55, 55, 70),
			}, Holder)
			Corner(Bar, 3)

			local Fill = Create("Frame", {
				Size = UDim2.new((default - min) / (max - min), 0, 1, 0),
				BackgroundColor3 = Theme.Accent,
			}, Bar)
			Corner(Fill, 3)

			local dragging = false
			local function Update(input)
				local rel = math.clamp((input.Position.X - Bar.AbsolutePosition.X) / Bar.AbsoluteSize.X, 0, 1)
				local value = math.floor(min + (max - min) * rel + 0.5)
				Fill.Size = UDim2.new(rel, 0, 1, 0)
				ValueLabel.Text = tostring(value)
				callback(value)
			end
			Bar.InputBegan:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = true; Update(input) end
			end)
			UserInputService.InputChanged:Connect(function(input)
				if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then Update(input) end
			end)
			UserInputService.InputEnded:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
			end)
		end

		-- ---------- ДРОПДАУН ----------
		function Tab:AddDropdown(text, options, default, callback)
			options = options or {}
			callback = callback or function() end
			local open = false
			local current = default or options[1]

			local Holder = Create("TextButton", {
				Size = UDim2.new(1, 0, 0, 38),
				BackgroundColor3 = Theme.Secondary,
				Text = "",
				AutoButtonColor = false,
			}, Page)
			Corner(Holder, 8)

			Create("TextLabel", {
				Size = UDim2.new(1, -70, 1, 0),
				Position = UDim2.new(0, 12, 0, 0),
				BackgroundTransparency = 1,
				Text = text .. ": " .. tostring(current),
				TextColor3 = Theme.Text,
				TextSize = 14,
				Font = Enum.Font.GothamMedium,
				TextXAlignment = Enum.TextXAlignment.Left,
			}, Holder).Name = "DropLabel"

			local Arrow = Create("ImageLabel", {
				Size = UDim2.fromOffset(16, 16),
				Position = UDim2.new(1, -30, 0.5, -8),
				BackgroundTransparency = 1,
				Image = GetIcon("chevron-small-down"),
				ImageColor3 = Theme.SubText,
			}, Holder)

			local List = Create("Frame", {
				Size = UDim2.new(1, 0, 0, 0),
				BackgroundColor3 = Theme.Background,
				Visible = false,
				ClipsDescendants = true,
			}, Page)
			Corner(List, 8)
			Create("UIListLayout", { Padding = UDim.new(0, 2), SortOrder = Enum.SortOrder.LayoutOrder }, List)
			Create("UIPadding", { PaddingTop = UDim.new(0, 4), PaddingBottom = UDim.new(0, 4) }, List)

			for _, opt in ipairs(options) do
				local OptBtn = Create("TextButton", {
					Size = UDim2.new(1, -8, 0, 30),
					Position = UDim2.new(0, 4, 0, 0),
					BackgroundColor3 = Theme.Secondary,
					Text = tostring(opt),
					TextColor3 = Theme.Text,
					TextSize = 13,
					Font = Enum.Font.Gotham,
					AutoButtonColor = false,
				}, List)
				Corner(OptBtn, 6)
				OptBtn.MouseEnter:Connect(function() Tween(OptBtn, 0.1, { BackgroundColor3 = Theme.Accent }) end)
				OptBtn.MouseLeave:Connect(function() Tween(OptBtn, 0.1, { BackgroundColor3 = Theme.Secondary }) end)
				OptBtn.MouseButton1Click:Connect(function()
					current = opt
					Holder:FindFirstChild("DropLabel").Text = text .. ": " .. tostring(opt)
					open = false
					Tween(List, 0.25, { Size = UDim2.new(1, 0, 0, 0) })
					task.wait(0.25)
					List.Visible = false
					callback(opt)
				end)
			end

			Holder.MouseButton1Click:Connect(function()
				open = not open
				List.Visible = true
				local h = #options * 32 + 8
				Tween(List, 0.3, { Size = open and UDim2.new(1, 0, 0, h) or UDim2.new(1, 0, 0, 0) })
				Tween(Arrow, 0.25, { Rotation = open and 180 or 0 })
				if not open then task.delay(0.3, function() if not open then List.Visible = false end end) end
			end)
		end

		-- ---------- ТЕКСТОВОЕ ПОЛЕ ----------
		function Tab:AddInput(text, placeholder, callback)
			callback = callback or function() end
			local Holder = Create("Frame", {
				Size = UDim2.new(1, 0, 0, 38),
				BackgroundColor3 = Theme.Secondary,
			}, Page)
			Corner(Holder, 8)

			Create("ImageLabel", {
				Size = UDim2.fromOffset(16, 16),
				Position = UDim2.new(0, 12, 0.5, -8),
				BackgroundTransparency = 1,
				Image = GetIcon("pencil"),
				ImageColor3 = Theme.SubText,
			}, Holder)

			local Box = Create("TextBox", {
				Size = UDim2.new(1, -50, 1, 0),
				Position = UDim2.new(0, 38, 0, 0),
				BackgroundTransparency = 1,
				Text = "",
				PlaceholderText = placeholder or "...",
				PlaceholderColor3 = Theme.SubText,
				TextColor3 = Theme.Text,
				TextSize = 14,

			Box.FocusLost:Connect(function(enter)

		end

		-- ---------- КЛАВИША ----------
		function Tab:AddKeybind(text, defaultKey, callback)
			callback = callback or function() end
			local key = defaultKey or Enum.KeyCode.RightShift

			local Holder = Create("TextButton", {
				Size = UDim2.new(1, 0, 0, 38),
				BackgroundColor3 = Theme.Secondary,
				Text = "",
				AutoButtonColor = false,
			}, Page)
			Corner(Holder, 8)

			Create("TextLabel", {
				Size = UDim2.new(1, -110, 1, 0),
				Position = UDim2.new(0, 12, 0, 0),
				BackgroundTransparency = 1,
				Text = text,
				TextColor3 = Theme.Text,
				TextSize = 14,
				Font = Enum.Font.GothamMedium,
				TextXAlignment = Enum.TextXAlignment.Left,
			}, Holder)

			local KeyLabel = Create("TextLabel", {
				Size = UDim2.new(0, 90, 0, 26),
				Position = UDim2.new(1, -98, 0.5, -13),
				BackgroundColor3 = Theme.Background,
				Text = key.Name,
				TextColor3 = Theme.Accent,
				TextSize = 12,
				Font = Enum.Font.GothamBold,
			}, Holder)
			Corner(KeyLabel, 6)

			local listening = false
			Holder.MouseButton1Click:Connect(function()
				listening = true
				KeyLabel.Text = "..."
			end)
			UserInputService.InputBegan:Connect(function(input, processed)
				if listening and input.UserInputType == Enum.UserInputType.Keyboard then
					key = input.KeyCode
					KeyLabel.Text = key.Name
					listening = false
				elseif not processed and input.KeyCode == key then
					callback(key)
				end
			end)
		end

		return Tab
	end

	return Window
end

return OxLib
