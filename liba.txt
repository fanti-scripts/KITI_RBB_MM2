    end;

        local path: StringValue = __get_path(path);
        return path.Value;
    end;

    getgenv().isfile = function(path)
        local success, message = pcall(function()
            return __get_path(path);
        end);
        if success and not message:IsA("Folder") then
            return true;
        end;
        return false;
    end;

    getgenv().isfolder = function(path)
        local success, message = pcall(function()
            return __get_path(path);
        end);
        if success and message:IsA("Folder") then
            return true;
        end;
        return false;
    end;

    getgenv().writefile = function(path, content)
        local main = __get_path_c(path);
        local block = BaseWorkspace;
        for i, v in next, main do
            local item = block:FindFirstChild(v);
            if not item then
                local c = Instance.new('StringValue', block);
                c.Name = tostring(v);
                c.Value = content;
            else
                if item:IsA('StringValue') and tostring(item) == v then
                    item.Name = tostring(v);
                    item.Value = content;
                end;
                block = item;
            end;
        end;
    end;

    getgenv().listfiles = function(path)
        local fold = __get_path(path);
        local pa = {};
        for i, v in next, fold:GetChildren() do
            if v:IsA('StringValue') then
                table.insert(pa, path .. '/' .. tostring(v));
            end;
        end;
        return pa;
    end;

    getgenv().makefolder = function(path)
        local main = __get_path_c(path);
        local block = BaseWorkspace;
        for i, v in next, main do
            local item = block:FindFirstChild(v);
            if not item then
                local c = Instance.new('Folder', block);
                c.Name = tostring(v);
            else
                block = item;
            end;
        end;
    end;

    getgenv().delfile = function(path)
        local main = __get_path_c(path);
        local block = BaseWorkspace;
        for i, v in next, main do
            local item = block:FindFirstChild(v);
            if item and item:IsA('StringValue') then
                item:Destroy();
            else
                block = item;
            end;
        end;
    end;
end;

function Blasted.RandomStr(len)
    local str = "";
    for i = 1, len do
        local idx = math.random(1, #Blasted.Ascii);
        str = str .. Blasted.Ascii:sub(idx, idx);
    end;
    return str;
end;

function Blasted.Rounding(num, decimalPlaces)
    local mult = 10 ^ (decimalPlaces or 0);
    return math.floor(num * mult + 0.5) / mult;
end;

function Blasted:IsMouseOverFrame(Frame)
    local AbsPos, AbsSize = Frame.AbsolutePosition, Frame.AbsoluteSize;
    local M = Mouse;
    return (M.X >= AbsPos.X and M.X <= AbsPos.X + AbsSize.X) and (M.Y >= AbsPos.Y and M.Y <= AbsPos.Y + AbsSize.Y);
end;

function Blasted:AddDragBlacklist(Frame)
    Frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.MouseMovement then
            if not table.find(Blasted.DragBlacklist, Frame) then
                table.insert(Blasted.DragBlacklist, Frame);
            end;
        end;
    end);
    Frame.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.MouseMovement then
            local finder = table.find(Blasted.DragBlacklist, Frame);
            if finder then
                table.remove(Blasted.DragBlacklist, finder);
            end;
        end;
    end);
end;

-- Drag System --
function Blasted:Drag(InputFrame, MoveFrame, Speed)
    local dragging, dragStart, startPos;

    InputFrame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if #Blasted.DragBlacklist > 0 then return end;
            dragging = true;
            dragStart = input.Position;
            startPos = MoveFrame.Position;

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false;
                end;
            end);
        end;
    end);

    InputFrame.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            if dragging then
                local delta = input.Position - dragStart;
                Blasted:CreateAnimation(MoveFrame, Speed or 0.1, Enum.EasingStyle.Quad, {
                    Position = UDim2.new(
                        startPos.X.Scale,
                        startPos.X.Offset + delta.X,
                        startPos.Y.Scale,
                        startPos.Y.Offset + delta.Y
                    )
                });
            end;
        end;
    end);
end;

    Signal.__Value = DefaultValue;
    Signal.__Event = Instance.new("BindableEvent");

    function Signal:GetValue()
        return Signal.__Value;
    end;

    function Signal:SetValue(f)
        Signal.__Value = f;
        Signal.__Event:Fire(f);
    end;

        Blasted:AddSignal(connection);

end;

-- Shadow System --
function Blasted:CreateShadow(Parent, RollingEffect)
    local Shadows = {};
    local Thicknesses = { 6, 5, 4, 3 };
    local Transparencies = { 0.9, 0.85, 0.8, 0.75 };

    for i = 1, 4 do
        local UIShadow = Instance.new("UIStroke");
        UIShadow.Name = "BlastedShadow" .. tostring(i);
        UIShadow.Color = Color3.fromRGB(0, 0, 0);
        UIShadow.Thickness = Thicknesses[i];
        UIShadow.Transparency = Transparencies[i];
        UIShadow.ApplyStrokeMode = Enum.ApplyStrokeMode.Border;
        UIShadow.Parent = Parent;
        table.insert(Shadows, UIShadow);

        if RollingEffect then
            local Gradient = Instance.new("UIGradient");
            Gradient.Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0, 0.4),
                NumberSequenceKeypoint.new(0.5, 0),
                NumberSequenceKeypoint.new(1, 0.4),
            });
            Gradient.Parent = UIShadow;
        end;
    end;

    local ShadowModule = {};
    function ShadowModule:Render(value)
        for _, Shadow in next, Shadows do
            if value then
                Blasted:CreateAnimation(Shadow, 0.3, Enum.EasingStyle.Quad, { Transparency = 0.85 });
            else
                Blasted:CreateAnimation(Shadow, 0.3, Enum.EasingStyle.Quad, { Transparency = 1 });
            end;
        end;

        if RollingEffect then
            if value then
                task.spawn(function()
                    while ShadowModule.__Rolling do
                        for _, Shadow in next, Shadows do
                            local Gradient = Shadow:FindFirstChildOfClass("UIGradient");
                            if Gradient then
                                Gradient.Rotation = Gradient.Rotation + 20;
                            end;
                        end;
                        task.wait(0.025);
                    end;
                end);
            end;
        end;
    end;

    return ShadowModule;
end;

-- Blur Module --
function Blasted:CreateBlurModule(Frame, Signal)
    local Part = Instance.new("Part");
    Part.Name = "BlastedBlurPart";
    Part.Anchored = true;
    Part.Size = Vector3.new(0.01, 0.01, 0.01);
    Part.Material = Enum.Material.Glass;
    Part.Transparency = 1;
    Part.Reflectance = 1;
    Part.CanCollide = false;
    Part.Parent = CurrentCamera;

    local BlockMesh = Instance.new("BlockMesh");
    BlockMesh.Parent = Part;

    local BlurEffect = Instance.new("DepthOfFieldEffect");
    BlurEffect.Name = "BlastedBlur";
    BlurEffect.FarIntensity = 0;
    BlurEffect.FocusDistance = 0;
    BlurEffect.InFocusRadius = 1000;
    BlurEffect.NearIntensity = 1;
    BlurEffect.Parent = Lighting;

    local function GetCalculatePosition(ScreenPoint)
        local Ray = CurrentCamera:ViewportPointToRay(ScreenPoint.X, ScreenPoint.Y);
        local Plane = Ray.Origin + Ray.Direction * (CurrentCamera.CFrame.Position.Z / Ray.Direction.Z * -1);
        return Plane;
    end;

    local function UpdateBlur()
        if not Frame or not Frame.Parent then return end;

        local Corners = {
            Frame.AbsolutePosition,
            Frame.AbsolutePosition + Vector2.new(Frame.AbsoluteSize.X, 0),
            Frame.AbsolutePosition + Frame.AbsoluteSize,
            Frame.AbsolutePosition + Vector2.new(0, Frame.AbsoluteSize.Y),
        };

        local WorldCorners = {};
        for _, Corner in next, Corners do
            table.insert(WorldCorners, GetCalculatePosition(Corner));
        end;

        local Center = (WorldCorners[1] + WorldCorners[3]) / 2;
        local Size = (WorldCorners[3] - WorldCorners[1]).Magnitude;

        Part.CFrame = CFrame.new(Center) * CFrame.new(0, 0, -0.5);
        Part.Size = Vector3.new(Size, Size, 0.01);
    end;

    local Connections = {};
    table.insert(Connections, RunService.RenderStepped:Connect(UpdateBlur));
    table.insert(Connections, Frame:GetPropertyChangedSignal("AbsolutePosition"):Connect(UpdateBlur));
    table.insert(Connections, Frame:GetPropertyChangedSignal("AbsoluteSize"):Connect(UpdateBlur));

    task.spawn(function()
        while Signal and Signal:GetValue() do
            UpdateBlur();
            task.wait(0.1);
        end;
    end);

    local BlurModule = {};
    function BlurModule:Render(value)
        if value then
            Part.Transparency = 1;
            BlurEffect.NearIntensity = 1;
        else
            Part.Transparency = 0;
            BlurEffect.NearIntensity = 0;
        end;
    end;

    function BlurModule:Destroy()
        for _, conn in next, Connections do
            conn:Disconnect();
        end;
        Part:Destroy();
        BlurEffect:Destroy();
    end;

    return BlurModule;
end;

-- Encryption System --
local Encryption = {};
Encryption.__index = Encryption;

function Encryption.new(data)
    local bytes = {};
    local encrypt_seed = ((#data + 3782) % 111) + 1;
    string.gsub(data, '.', function(dt)
        table.insert(bytes, tostring(dt:byte() + encrypt_seed));
    end);
    local concatbyte = table.concat(bytes, '?');
    table.clear(bytes);
    return "{" .. tostring(encrypt_seed + 72667) .. "}?" .. concatbyte;
end;

function Encryption.reverse(data)
    local main_data = string.split(data, '?');
    local seed_str = main_data[1]:gsub('{', ''):gsub('}', '');
    local seed = tonumber(seed_str);
    local ks = {};
    local real_seed = seed - 72667;
    for i, v in next, main_data do
        if i > 1 then
            local fake_byte = tonumber(v);
            table.insert(ks, string.char(fake_byte - real_seed));
        end;
    end;
    local data = table.concat(ks);
    table.clear(ks);
    return data;
end;

-- Base64 --
do
    local b = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/';
    function Blasted.Base64Encode(data)
        return ((data:gsub('.', function(x)
            local r, b2 = '', x:byte();
            for i = 8, 1, -1 do
                r = r .. (b2 % 2 ^ i - b2 % 2 ^ (i - 1) > 0 and '1' or '0');
            end;
            return r;
        end) .. '0000'):gsub('%d%d%d?%d?%d?%d?', function(x)
            if #x ~= 6 then return '' end;
            local c = 0;
            for i = 1, 6 do
                c = c + (x:sub(i, i) == '1' and 2 ^ (6 - i) or 0);
            end;
            return b:sub(c + 1, c + 1);
        end) .. ({ '', '==', '=' })[#data % 3 + 1]);
    end;

    function Blasted.Base64Decode(data)
        data = string.gsub(data, '[^' .. b .. ']', '');
        return (data:gsub('.', function(x)
            if x == '=' then return '' end;
            local r, f = '', (b:find(x) - 1);
            for i = 6, 1, -1 do
                r = r .. (f % 2 ^ i - f % 2 ^ (i - 1) > 0 and '1' or '0');
            end;
            return r;
        end):gsub('%d%d%d?%d?%d?%d?%d?%d?', function(x)
            if #x ~= 8 then return '' end;
            local c = 0;
            for i = 1, 8 do
                c = c + (x:sub(i, i) == '1' and 2 ^ (8 - i) or 0);
            end;
            return string.char(c);
        end));
    end;
end;

function Blasted:GetIcon(name, system)
    if system == "lucide" or system == nil then
        return Blasted.Lucide['lucide-' .. name] or Blasted.Lucide[name] or "";
    elseif system == "fontawesome" then
        return Blasted.FontAwesome['fa-' .. name] or Blasted.FontAwesome[name] or "";
    elseif system == "builder" then
        return name;
    end;
    return "";
end;

function Blasted:SetIcon(name, value)
    if name:find("lucide") then
        Blasted.Lucide[name] = value;
    else
        Blasted.FontAwesome[name] = value;
    end;
end;

-- CreateResponse (for method binding) --
function Blasted:CreateResponse(args)
    local response = {};
    for k, v in pairs(args) do
        if type(v) == "function" then
            response[k] = function(_, ...)
                return v(...);
            end;
        else
            response[k] = v;
        end;
    end;
    return response;
end;

-- Protector (RichText obfuscation) --
function Blasted:ProtectText(Label, Text)
    local richText = "";
    for i = 1, #Text do
        local char = Text:sub(i, i);
        local fontIdx = math.random(1, 5);
        local fonts = { "GothamBold", "GothamMedium", "Gotham", "GothamBlack", "GothamBook" };
        richText = richText .. '<font family="rbxasset://fonts/families/' .. fonts[fontIdx] .. '.json">' .. char .. '</font>';
    end;
    Label.RichText = true;
    Label.Text = richText;
end;

Blasted.KeyEnum = {
    LeftControl = "LCtrl", RightControl = "RCtrl",
    LeftShift = "LShift", RightShift = "RShift",
    LeftAlt = "LAlt", RightAlt = "RAlt",
    Unknown = "...",
};
Blasted.EnumReverse = {};
for k, v in pairs(Blasted.KeyEnum) do
    Blasted.EnumReverse[v] = k;
end;

function Blasted:KeyCodeToStr(K)
    if typeof(K) == "EnumItem" then
        return Blasted.KeyEnum[K.Name] or K.Name;
    elseif typeof(K) == "string" then
        return K;
    end;
    return "Unknown";
end;

function Blasted:StrToKeyCode(str)
    return Blasted.EnumReverse[str] or str;
end;

    function ConfigSystem:SaveData(Name)
        Name = Name or ConfigSystem.CurrentConfig;
        local Data = ConfigSystem:GetData();
        local Encoded = HttpService:JSONEncode(Data);
        local Encrypted = Encryption.new(Encoded);
        local Base64Encoded = Blasted.Base64Encode(Encrypted);

        if not isfolder(ConfigSystem.Directory) then
            makefolder(ConfigSystem.Directory);
        end;

        writefile(ConfigSystem.Directory .. "/" .. Name .. ".blast", Base64Encoded);
    end;

    function ConfigSystem:LoadData(Name)
        Name = Name or ConfigSystem.CurrentConfig;
        local path = ConfigSystem.Directory .. "/" .. Name .. ".blast";
        if not isfile(path) then return false end;

        local Base64Encoded = readfile(path);
        local Encrypted = Blasted.Base64Decode(Base64Encoded);
        local Decoded = Encryption.reverse(Encrypted);
        local Data = HttpService:JSONDecode(Decoded);

        for FlagId, Value in next, Data do
            if Blasted.WindowFlags[ScreenGui][FlagId] and Blasted.WindowFlags[ScreenGui][FlagId].SetValue then
                Blasted.WindowFlags[ScreenGui][FlagId]:SetValue(Value);
            end;
        end;
        return true;
    end;

    function ConfigSystem:GetConfigs()
        if not isfolder(ConfigSystem.Directory) then
            makefolder(ConfigSystem.Directory);
        end;
        local files = listfiles(ConfigSystem.Directory);
        local configs = {};
        for _, file in next, files do
            if file:find("%.blast$") then
                table.insert(configs, file:match("([^/]+)%.blast$"));
            end;
        end;
        return configs;
    end;

    function ConfigSystem:DeleteConfig(Name)
        local path = ConfigSystem.Directory .. "/" .. Name .. ".blast";
        if isfile(path) then
            delfile(path);
        end;
    end;

                    local scale = (_Value - Config.Min) / (Config.Max - Config.Min);
                    scale = math.clamp(scale, 0, 1);
                    Blasted:CreateAnimation(boxli, 0.1, Enum.EasingStyle.Quad, {
                        Size = UDim2.new(scale, 0, 1, 0),
                    });
                    ValueText.Text = Blasted.Rounding(_Value, Config.Round) .. Config.Type;
                end;

                    return _Value;
                end;

                    _Value = math.clamp(v, Config.Min, Config.Max);
                    UpdateSlider();
                    Config.Callback(_Value);
                end;

                    SliderName.Text = newName;
                end;

                    Blasted.WindowFlags[ScreenGui][Config.Flag .. "Slider"] = Slider;

                            local absPos = ValueFrame.AbsolutePosition.X;
                            local absSize = ValueFrame.AbsoluteSize.X;
                            local scale = math.clamp((inp.Position.X - absPos) / absSize, 0, 1);

                            if input.UserInputState == Enum.UserInputState.End then
                                IsHolding = false;
                                conn:Disconnect();
                            end;
                        end);

                end);

                    if IsHolding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                        local absPos = ValueFrame.AbsolutePosition.X;
                        local absSize = ValueFrame.AbsoluteSize.X;
                        local scale = math.clamp((input.Position.X - absPos) / absSize, 0, 1);
                        _Value = Config.Min + (Config.Max - Config.Min) * scale;
                        UpdateSlider();
                        Slider.Signal:SetValue(_Value);
                        Config.Callback(_Value);
                    end;
                end);

                if Event then
                    Event.Event:Connect(function(visible)
                        SliderFrame.Visible = visible;
                    end);
                end;

                if SearchAPI then
                    SearchAPI.Memory(Config.Name);
                end;

                UpdateSlider();
                return Slider;
            end;

                    if input.KeyCode ~= Enum.KeyCode.Unknown then
                        key = input.KeyCode;
                    elseif input.UserInputType == Enum.UserInputType.MouseButton1 then
                        key = "MouseLeft";
                    elseif input.UserInputType == Enum.UserInputType.MouseButton2 then
                        key = "MouseRight";
                    end;

                    if key then
                        local keyStr = typeof(key) == "EnumItem" and key.Name or tostring(key);
                        local blocked = false;
                        for _, b in next, Config.Blacklist do
                            local bStr = typeof(b) == "EnumItem" and b.Name or tostring(b);

                -- Search box --
                local SearchFrame = Instance.new("Frame");
                SearchFrame.Name = "SearchFrame";
                SearchFrame.Size = UDim2.new(1, -6, 0, 18);
                SearchFrame.Position = UDim2.new(0, 3, 0, 3);
                SearchFrame.BackgroundColor3 = Blasted.CurrentTheme.WindowBg;
                SearchFrame.BorderSizePixel = 0;
                SearchFrame.ZIndex = zIdx + 1;
                SearchFrame.Parent = ElementsFrame;

                local SFCorner = Instance.new("UICorner");
                SFCorner.CornerRadius = UDim.new(0, 3);
                SFCorner.Parent = SearchFrame;

                local SearchBox = Instance.new("TextBox");
                SearchBox.Size = UDim2.new(1, -8, 1, 0);
                SearchBox.Position = UDim2.new(0, 4, 0, 0);
                SearchBox.BackgroundTransparency = 1;
                SearchBox.Text = "";
                SearchBox.PlaceholderText = "Start typing...";
                SearchBox.PlaceholderColor3 = Blasted.CurrentTheme.TextDark;
                SearchBox.TextColor3 = Blasted.CurrentTheme.Text;
                SearchBox.Font = Enum.Font.Gotham;
                SearchBox.TextSize = 10;
                SearchBox.ClearTextOnFocus = false;
                SearchBox.ZIndex = zIdx + 2;
                SearchBox.Parent = SearchFrame;

                local LBScroll = Instance.new("ScrollingFrame");
                LBScroll.Name = "LBScroll";
                LBScroll.Size = UDim2.new(1, -6, 1, -24);
                LBScroll.Position = UDim2.new(0, 3, 0, 24);
                LBScroll.BackgroundTransparency = 1;
                LBScroll.ScrollBarThickness = 0;
                LBScroll.AutomaticCanvasSize = Enum.AutomaticSize.Y;
                LBScroll.ZIndex = zIdx + 1;
                LBScroll.Parent = ElementsFrame;

                local LBLayout = Instance.new("UIListLayout");
                LBLayout.Padding = UDim.new(0, 2);
                LBLayout.ZIndex = zIdx + 1;
                LBLayout.Parent = LBScroll;

                local _Value = Config.Default;
                local Buttons = {};

                local function Generate()
                    for _, btn in next, Buttons do
                        btn:Destroy();
                    end;
                    Buttons = {};

                    for _, val in next, Config.Values do
                        local Item = Instance.new("TextButton");
                        Item.Name = tostring(val);
                        Item.Size = UDim2.new(1, 0, 0, 15);
                        Item.BackgroundColor3 = Blasted.CurrentTheme.WindowBg;
                        Item.BackgroundTransparency = 0.5;
                        Item.Text = "  " .. tostring(val);
                        Item.TextColor3 = Blasted.CurrentTheme.Text;
                        Item.Font = Enum.Font.GothamMedium;
                        Item.TextSize = 10;
                        Item.TextXAlignment = Enum.TextXAlignment.Left;
                        Item.ZIndex = zIdx + 2;
                        Item.Parent = LBScroll;

                        local ItemCorner = Instance.new("UICorner");
                        ItemCorner.CornerRadius = UDim.new(0, 2);
                        ItemCorner.Parent = Item;

                        table.insert(Buttons, Item);

                        if Config.Multi and _Value and _Value[val] then
                            Item.BackgroundTransparency = 0;
                            Item.BackgroundColor3 = Blasted.CurrentTheme.Main;
                        elseif not Config.Multi and _Value == val then
                            Item.BackgroundTransparency = 0;
                            Item.BackgroundColor3 = Blasted.CurrentTheme.Main;
                        end;

                        Item.MouseButton1Click:Connect(function()
                            if Config.Multi then
                                _Value = _Value or {};
                                _Value[val] = not _Value[val];
                                Item.BackgroundTransparency = _Value[val] and 0 or 0.5;
                                Item.BackgroundColor3 = _Value[val] and Blasted.CurrentTheme.Main or Blasted.CurrentTheme.WindowBg;
                            else
                                for _, b in next, Buttons do
                                    b.BackgroundTransparency = 0.5;
                                    b.BackgroundColor3 = Blasted.CurrentTheme.WindowBg;
                                end;
                                _Value = val;
                                Item.BackgroundTransparency = 0;
                                Item.BackgroundColor3 = Blasted.CurrentTheme.Main;
                            end;
                            Config.Callback(_Value);
                        end);
                    end;
                end;

                -- Search functionality --
                local SearchDelay = tick();
                SearchBox:GetPropertyChangedSignal("Text"):Connect(function()
                    SearchDelay = tick();
                    local searchText = SearchBox.Text:lower();

                    if searchText == "" then
                        for _, btn in next, Buttons do
                            btn.Visible = true;
                        end;
                        return;
                    end;

                    task.delay(0.25, function()
                        if (tick() - SearchDelay) > 0.25 then
                            for _, btn in next, Buttons do
                                btn.Visible = string.find(btn.Text:lower(), searchText, 1, true) ~= nil;
                            end;
                        end;
                    end);
                end);

                local ListBox = {};
                ListBox.GetValue = function()
                    return _Value;
                end;
                ListBox.SetValue = function(_, v)
                    _Value = v;
                    Generate();
                end;
                ListBox.SetValues = function(_, t)
                    Config.Values = t;
                    Generate();
                end;
                ListBox.Refresh = Generate;
                ListBox.Signal = Blasted:CreateSignal(_Value);

                if Config.Flag then
                    Blasted.WindowFlags[ScreenGui][Config.Flag .. "ListBox"] = ListBox;
                end;

                if Event then
                    Event.Event:Connect(function(visible)
                        LBFrame.Visible = visible;
                    end);
                end;

                if SearchAPI then
                    SearchAPI.Memory(Config.Name);
                end;

                Generate();
                return ListBox;
            end;

            return Elements;
        end;

-- LOADER ANIMATION
-- ═══════════════════════════════════════════════════════════════════════════════

function Blasted:Loader(Config)
    Config = Blasted:ProcessParams(Config, {
        Name = "BLASTED",
        Duration = 3.5,
        Scale = 3,
    });

    local LoaderScreen = Instance.new("ScreenGui");
    LoaderScreen.Name = Blasted.RandomStr(12);
    LoaderScreen.ResetOnSpawn = false;
    LoaderScreen.ZIndexBehavior = Enum.ZIndexBehavior.Global;
    LoaderScreen.IgnoreGuiInset = true;
    ProtectGui(LoaderScreen);
    LoaderScreen.Parent = CoreGui;

    local center = Instance.new("Frame");
    center.Size = UDim2.new(1, 0, 1, 0);
    center.BackgroundTransparency = 1;
    center.Parent = LoaderScreen;

    local texts = Instance.new("Frame");
    texts.Name = "texts";
    texts.Size = UDim2.new(1, 0, 0, 200);
    texts.Position = UDim2.new(0.5, 0, 0.5, 0);
    texts.AnchorPoint = Vector2.new(0.5, 0.5);
    texts.BackgroundTransparency = 1;
    texts.Parent = LoaderScreen;

    local textsLayout = Instance.new("UIListLayout");
    textsLayout.FillDirection = Enum.FillDirection.Horizontal;
    textsLayout.HorizontalAlignment = Enum.HorizontalAlignment.Center;
    textsLayout.VerticalAlignment = Enum.VerticalAlignment.Center;
    textsLayout.Padding = UDim.new(0, Config.Scale * 5);
    textsLayout.Parent = texts;

    local BlackFrame = Instance.new("Frame");
    BlackFrame.Size = UDim2.new(1, 0, 1, 0);
    BlackFrame.BackgroundColor3 = Color3.new(0, 0, 0);
    BlackFrame.BackgroundTransparency = 1;
    BlackFrame.BorderSizePixel = 0;
    BlackFrame.Parent = LoaderScreen;

    -- Blur --
    local Blur = Instance.new("BlurEffect");
    Blur.Size = 0;
    Blur.Parent = Lighting;
    Blasted:CreateAnimation(Blur, 1, nil, { Size = 60 });

    Blasted:CreateAnimation(BlackFrame, 0.5, nil, { BackgroundTransparency = 0.7 });

    -- Animate each character --
    local chars = {};
    for i = 1, #Config.Name do
        local char = Config.Name:sub(i, i);

        local charFrame = Instance.new("Frame");
        charFrame.Size = UDim2.new(0, 0, 0, 50);
        charFrame.BackgroundTransparency = 1;
        charFrame.Parent = texts;

        local charLabel = Instance.new("TextLabel");
        charLabel.Size = UDim2.new(1, 0, 1, 0);
        charLabel.BackgroundTransparency = 1;
        charLabel.Text = char;
        charLabel.TextColor3 = Blasted.CurrentTheme.Main;
        charLabel.Font = Enum.Font.GothamBold;
        charLabel.TextSize = 50;
        charLabel.Parent = charFrame;

        local gradient = Instance.new("UIGradient");
        gradient.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Blasted.CurrentTheme.Main),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(100, 0, 0)),
        });
        gradient.Rotation = 88;
        gradient.Parent = charLabel;

        local scale = Instance.new("UIScale");
        scale.Scale = Config.Scale;
        scale.Parent = charLabel;

        table.insert(chars, { frame = charFrame, label = charLabel, scale = scale });

        task.defer(function()
            charFrame.Position = UDim2.new(0, 0, 1, 0);
            Blasted:CreateAnimation(charFrame, 0.3 + i * 0.1, Enum.EasingStyle.Quint, {
                Position = UDim2.new(0, 0, 0, 0),
            });
            charLabel.TextTransparency = 1;
            Blasted:CreateAnimation(charLabel, 0.3 + i * 0.1, nil, {
                TextTransparency = 0,
            });
        end);
    end;

    task.wait(Config.Duration);

    -- Fade out --
    for _, char in next, chars do
        Blasted:CreateAnimation(char.label, 0.3, nil, { TextTransparency = 1 });
    end;

    Blasted:CreateAnimation(Blur, 1.5, nil, { Size = 0 });
    Blasted:CreateAnimation(BlackFrame, 1.5, nil, { BackgroundTransparency = 1 });

    task.wait(1.5);

    Blur:Destroy();
    LoaderScreen:Destroy();
end;

-- UNLOAD
-- ═══════════════════════════════════════════════════════════════════════════════

function Blasted:Unload()
    Blasted.UnloadEnabled = true;
    for _, signal in next, Blasted.GlobalSignals do
        if signal and signal.Connected then
            signal:Disconnect();
        end;
    end;
    for _, screen in next, Blasted.Windows do
        if screen and screen.Parent then
            screen:Destroy();
        end;
    end;
    table.clear(Blasted.Windows);
    table.clear(Blasted.GlobalSignals);
    table.clear(Blasted.NameRegistry);
    table.clear(Blasted.Flags);
    Blasted.__NOTIFIER_CACHE = nil;
    Blasted.__WATERMARK_CACHE = nil;
    Blasted.__INDICATOR_CACHE = nil;
    Blasted.__LOGGER_CACHE = nil;
end;

-- Expose to getgenv for easy access --
getgenv().Blasted = Blasted;

return Blasted;
