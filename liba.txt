local Real = {}
Real.__index = Real

--// Config (стиль сайта)
local THEME = {
    Background   = Color3.fromRGB(10, 10, 12),   -- почти чёрный как у Real
    Surface      = Color3.fromRGB(18, 18, 22),
    SurfaceLight = Color3.fromRGB(26, 26, 32),
    Border       = Color3.fromRGB(38, 38, 46),
    Accent       = Color3.fromRGB(155, 126, 240), -- #9b7ef0 фирменный фиолетовый
    AccentDark   = Color3.fromRGB(120, 95, 200),
    Text         = Color3.fromRGB(235, 235, 240),
    TextDim      = Color3.fromRGB(140, 140, 150),
    Font         = Enum.Font.GothamMedium,
    FontBold     = Enum.Font.GothamBold,
    Radius       = 12,
}

local function Create(class, props, children)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then inst[k] = v end
    end
    for _, c in ipairs(children or {}) do c.Parent = inst end
    inst.Parent = props.Parent
    return inst
end

local function Corner(r) return Create("UICorner", {CornerRadius = UDim.new(0, r or THEME.Radius)}) end
local function Stroke(color, t) return Create("UIStroke", {Color = color or THEME.Border, Thickness = t or 1}) end

--═══════════════════════════════════════════════
--  WINDOW
--═══════════════════════════════════════════════
function Real:CreateWindow(title)
    local self = setmetatable({}, Real)

    -- ScreenGui
    local gui = Create("ScreenGui", {
        Name = "RealUI",
        ResetOnSpawn = false,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
    }, nil, game:GetService("CoreGui"))

    -- Main frame
    local main = Create("Frame", {
        Size = UDim2.fromOffset(560, 380),
        Position = UDim2.new(0.5, -280, 0.5, -190),
        BackgroundColor3 = THEME.Background,
        Active = true,
        Draggable = true,
    }, {Corner(), Stroke()}, gui)

    -- Top bar (как nav на сайте)
    local topbar = Create("Frame", {
        Size = UDim2.new(1, 0, 0, 44),
        BackgroundColor3 = THEME.Surface,
    }, {Corner()}, main)

    Create("UICorner", {CornerRadius = UDim.new(0, THEME.Radius)}, {}, topbar)

    Create("TextLabel", {
        Size = UDim2.new(1, -100, 1, 0),
        Position = UDim2.fromOffset(16, 0),
        BackgroundTransparency = 1,
        Text = title or "Real",
        TextColor3 = THEME.Text,
        Font = THEME.FontBold,
        TextSize = 15,
        TextXAlignment = Enum.TextXAlignment.Left,
    }, {}, topbar)

    -- Logo dot (фиолетовый кружок как бренд Real)
    Create("Frame", {
        Size = UDim2.fromOffset(10, 10),
        Position = UDim2.new(1, -28, 0.5, -5),
        BackgroundColor3 = THEME.Accent,
    }, {Create("UICorner", {CornerRadius = UDim.new(1, 0)})}, topbar)

    -- Close btn
    local closeBtn = Create("TextButton", {
        Size = UDim2.fromOffset(28, 28),
        Position = UDim2.new(1, -64, 0.5, -14),
        BackgroundColor3 = THEME.SurfaceLight,
        Text = "✕",
        TextColor3 = THEME.TextDim,
        Font = THEME.Font,
        TextSize = 13,
        AutoButtonColor = true,
    }, {Create("UICorner", {CornerRadius = UDim.new(0, 8)})}, topbar)

    closeBtn.MouseButton1Click:Connect(function()
        gui:Destroy()
    end)

    -- Tab bar слева
    local tabHolder = Create("ScrollingFrame", {
        Size = UDim2.new(0, 130, 1, -44),
        Position = UDim2.fromOffset(0, 44),
        BackgroundTransparency = 1,
        ScrollBarThickness = 0,
        CanvasSize = UDim2.new(),
        AutomaticCanvasSize = Enum.AutomaticSize.Y,
    }, {
        Create("UIListLayout", {Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder}),
        Create("UIPadding", {PaddingTop = UDim.new(0, 10), PaddingLeft = UDim.new(0, 10)}),
    }, main)

    -- Page holder справа
    local pageHolder = Create("Frame", {
        Size = UDim2.new(1, -150, 1, -54),
        Position = UDim2.fromOffset(140, 54),
        BackgroundTransparency = 1,
    }, {}, main)

    self._gui = gui
    self._main = main
    self._tabHolder = tabHolder
    self._pageHolder = pageHolder
    self._tabs = {}

    function self:SelectTab(tab)
        for name, t in pairs(self._tabs) do
            t.button.BackgroundColor3 = (name == tab) and THEME.SurfaceLight or THEME.Surface
            t.button.TextColor3 = (name == tab) and THEME.Accent or THEME.TextDim
            t.page.Visible = (name == tab)
        end
    end

    --═══════════════════════════════════════════
    --  TAB
    --═══════════════════════════════════════════
    function self:AddTab(name)
        local btn = Create("TextButton", {
            Size = UDim2.new(1, 0, 0, 34),
            BackgroundColor3 = THEME.Surface,
            Text = "  " .. name,
            TextColor3 = THEME.TextDim,
            Font = THEME.Font,
            TextSize = 13,
            TextXAlignment = Enum.TextXAlignment.Left,
            AutoButtonColor = true,
        }, {Create("UICorner", {CornerRadius = UDim.new(0, 8)})}, tabHolder)

        local page = Create("ScrollingFrame", {
            Size = UDim2.new(1, 0, 1, 0),
            BackgroundTransparency = 1,
            ScrollBarThickness = 3,
            ScrollBarImageColor3 = THEME.Accent,
            CanvasSize = UDim2.new(),
            AutomaticCanvasSize = Enum.AutomaticSize.Y,
            Visible = false,
        }, {
            Create("UIListLayout", {Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder}),
            Create("UIPadding", {PaddingRight = UDim.new(0, 8)}),
        }, pageHolder)

        self._tabs[name] = {button = btn, page = page}
        btn.MouseButton1Click:Connect(function() self:SelectTab(name) end)

        local tab = {}

        -- Section header
        function tab:AddSection(text)
            Create("TextLabel", {
                Size = UDim2.new(1, 0, 0, 24),
                BackgroundTransparency = 1,
                Text = string.upper(text),
                TextColor3 = THEME.TextDim,
                Font = THEME.FontBold,
                TextSize = 11,
                TextXAlignment = Enum.TextXAlignment.Left,
            }, {}, page)
        end

        -- Button
        function tab:AddButton(text, callback)
            local b = Create("TextButton", {
                Size = UDim2.new(1, 0, 0, 36),
                BackgroundColor3 = THEME.Surface,
                Text = text,
                TextColor3 = THEME.Text,
                Font = THEME.Font,
                TextSize = 13,
                AutoButtonColor = true,
            }, {Corner(8), Stroke()}, page)

            b.MouseEnter:Connect(function() b.BackgroundColor3 = THEME.SurfaceLight end)
            b.MouseLeave:Connect(function() b.BackgroundColor3 = THEME.Surface end)
            b.MouseButton1Click:Connect(function()
                pcall(callback)
            end)
        end

        -- Toggle
        function tab:AddToggle(text, default, callback)
            local state = default or false
            local row = Create("TextButton", {
                Size = UDim2.new(1, 0, 0, 36),
                BackgroundColor3 = THEME.Surface,
                Text = "",
                AutoButtonColor = false,
            }, {Corner(8), Stroke()}, page)

            Create("TextLabel", {
                Size = UDim2.new(1, -60, 1, 0),
                Position = UDim2.fromOffset(14, 0),
                BackgroundTransparency = 1,
                Text = text,
                TextColor3 = THEME.Text,
                Font = THEME.Font,
                TextSize = 13,
                TextXAlignment = Enum.TextXAlignment.Left,
            }, {}, row)

            local pill = Create("Frame", {
                Size = UDim2.fromOffset(40, 20),
                Position = UDim2.new(1, -52, 0.5, -10),
                BackgroundColor3 = state and THEME.Accent or THEME.SurfaceLight,
            }, {Create("UICorner", {CornerRadius = UDim.new(1, 0)})}, row)

            local knob = Create("Frame", {
                Size = UDim2.fromOffset(14, 14),
                Position = state and UDim2.new(1, -17, 0.5, -7) or UDim2.fromOffset(3, 3),
                BackgroundColor3 = Color3.new(1, 1, 1),
            }, {Create("UICorner", {CornerRadius = UDim)})}, pill)

            row.MouseButton1Click:Connect(function()
                state = not state
                callback(state)
                -- анимация как плавные переходы на сайте
                local tween = game:GetService("TweenService")
                tween:Create(pill, TweenInfo.new(0.15), {BackgroundColor3 = state and THEME.Accent or THEME.SurfaceLight}):Play()
                tween:Create(knob, TweenInfo.new(0.15), {Position = state and UDim2.new(1, -17, 0.5, -7) or UDim2.fromOffset(3, 3)}):Play()
            end)
        end

        -- Slider
        function tab:AddSlider(text, min, max, default, callback)
            local value = default or min
            local container = Create("Frame", {
                Size = UDim2.new(1, 0, 0, 50),
                BackgroundColor3 = THEME.Surface,
            }, {Corner(8), Stroke()}, page)

            local label = Create("TextLabel", {
                Size = UDim2.new(1, -28, 0, 20),
                Position = UDim2.fromOffset(14, 6),
                BackgroundTransparency = 1,
                Text = text .. ": " .. tostring(value),
                TextColor3 = THEME.Text,
                Font = THEME.Font,
                TextSize = 13,
                TextXAlignment = Enum.TextXAlignment.Left,
            }, {}, container)

            local track = Create("Frame", {
                Size = UDim2.new(1, -28, 0, 4),
                Position = UDim2.new(0, 14, 0, 34),
                BackgroundColor3 = THEME.SurfaceLight,
            }, {Create("UICorner", {CornerRadius = UDim.new(1, 0)})}, container)

            local fill = Create("Frame", {
                Size = UDim2.new((value - min) / (max - min), 0, 1, 0),
                BackgroundColor3 = THEME.Accent,
            }, {Create("UICorner", {CornerRadius = UDim.new(1, 0)})}, track)

            local dragging = false
            local function update(input)
                local rel = math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
                value = math.floor(min + (max - min) * rel + 0.5)
                fill.Size = UDim2.new(rel, 0, 1, 0)
                label.Text = text .. ": " .. tostring(value)
                callback(value)
            end

            track.InputBegan:Connect(function(i)
                if i.UserInputType == Enum.UserInputType.MouseButton1 then
                    dragging = true; update(i)
                end
            end)
            game:GetService("UserInputService").InputChanged:Connect(function(i)
                if dragging and i.UserInputType == Enum.UserInputType.MouseMovement then update(i) end
            end)
            game:GetService("UserInputService").InputEnded:Connect(function(i)
                if i.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
            end)
        end

        return tab
    end

    -- Открыть первую вкладку
    task.defer(function()
        local first = next(self._tabs)
        if first then self:SelectTab(first) end
    end)

    return self
end

return Real
